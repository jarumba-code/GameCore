cmake_minimum_required(VERSION 3.22.1)
project(magpie_graphics LANGUAGES CXX)
add_library(magpie_graphics SHARED native_fsr.cpp
    ../../../../graphics/fsr1/src/image.cpp ../../../../graphics/fsr1/src/easu.cpp ../../../../graphics/fsr1/src/rcas.cpp)
target_include_directories(magpie_graphics PRIVATE ../../../../graphics/fsr1/include)
target_compile_features(magpie_graphics PRIVATE cxx_std_17)
target_compile_options(magpie_graphics PRIVATE -Wall -Wextra -Wpedantic)
if(ANDROID)
  find_library(android-lib android REQUIRED)
  find_library(vulkan_lib vulkan)
  target_link_libraries(magpie_graphics PRIVATE ${android-lib})
  if(vulkan_lib)
    target_compile_definitions(magpie_graphics PRIVATE MAGPIE_HAS_VULKAN=1)
    target_link_libraries(magpie_graphics PRIVATE ${vulkan_lib})
  endif()
endif()
