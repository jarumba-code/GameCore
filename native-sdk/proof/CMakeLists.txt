cmake_minimum_required(VERSION 3.22)
project(magpie_native_sdk_proof LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(MAGPIE_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)
add_library(magpie_core SHARED src/core.cpp)
target_include_directories(magpie_core PUBLIC include)
set_target_properties(magpie_core PROPERTIES OUTPUT_NAME magpie_core CXX_VISIBILITY_PRESET hidden)
add_library(magpie_fsr SHARED src/fsr.cpp
 ${MAGPIE_ROOT}/graphics/fsr1/src/image.cpp ${MAGPIE_ROOT}/graphics/fsr1/src/easu.cpp ${MAGPIE_ROOT}/graphics/fsr1/src/rcas.cpp)
target_include_directories(magpie_fsr PUBLIC include PRIVATE ${MAGPIE_ROOT}/graphics/fsr1/include)
target_link_libraries(magpie_fsr PRIVATE magpie_core)
set_target_properties(magpie_fsr PROPERTIES OUTPUT_NAME magpie_fsr CXX_VISIBILITY_PRESET hidden)
find_path(VULKAN_INCLUDE_DIR vulkan/vulkan.h REQUIRED)
add_library(magpie_vulkan SHARED src/vulkan_probe.cpp)
target_include_directories(magpie_vulkan PRIVATE include ${VULKAN_INCLUDE_DIR})
target_link_libraries(magpie_vulkan PRIVATE /system/lib64/libvulkan.so)
set_target_properties(magpie_vulkan PROPERTIES OUTPUT_NAME magpie_vulkan CXX_VISIBILITY_PRESET hidden)
add_library(magpie_gles SHARED src/gles_probe.cpp)
target_include_directories(magpie_gles PRIVATE include)
target_link_libraries(magpie_gles PRIVATE /system/lib64/libEGL.so /system/lib64/libGLESv2.so)
set_target_properties(magpie_gles PROPERTIES OUTPUT_NAME magpie_gles CXX_VISIBILITY_PRESET hidden)
add_executable(magpie_sdk_smoke tests/smoke.cpp)
target_include_directories(magpie_sdk_smoke PRIVATE include)
target_link_libraries(magpie_sdk_smoke PRIVATE magpie_core magpie_fsr magpie_vulkan magpie_gles)
enable_testing()
add_test(NAME magpie_sdk_shared_library_smoke COMMAND magpie_sdk_smoke)
